cmake_minimum_required(VERSION 3.20)
project(Kernoryx VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(kernoryx_power src/sysfs.cpp)
target_include_directories(kernoryx_power PUBLIC include)
target_compile_options(kernoryx_power PRIVATE -Wall -Wextra -Wpedantic)
add_executable(kernoryxd src/daemon.cpp)
target_link_libraries(kernoryxd PRIVATE kernoryx_power)
add_executable(kernoryx src/main.cpp)
include(CTest)
if(BUILD_TESTING)
  add_executable(policy-test tests/policy.cpp)
  target_link_libraries(policy-test PRIVATE kernoryx_power)
  add_test(NAME policy COMMAND policy-test)
  add_executable(controller-test tests/controller.cpp)
  target_link_libraries(controller-test PRIVATE kernoryx_power)
  add_test(NAME controller COMMAND controller-test)
endif()
install(TARGETS kernoryx RUNTIME DESTINATION bin)
install(TARGETS kernoryxd RUNTIME DESTINATION sbin)
