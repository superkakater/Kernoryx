cmake_minimum_required(VERSION 3.20)
project(Kernoryx VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
include(CTest)
add_compile_options(-Wall -Wextra -Wpedantic)
add_library(kernoryx_system src/system.cpp)
target_include_directories(kernoryx_system PUBLIC src)
add_library(kernoryx_core src/power.cpp src/process.cpp src/fan.cpp src/health.cpp)
target_link_libraries(kernoryx_core PUBLIC kernoryx_system)
add_executable(kernoryxd src/daemon.cpp)
target_link_libraries(kernoryxd PRIVATE kernoryx_core)
add_executable(kernoryx src/main.cpp)
target_compile_definitions(kernoryx PRIVATE
    KX_HELPER="${CMAKE_INSTALL_PREFIX}/libexec/kernoryx/maintenance.py"
    KX_BUILD_HELPER="${CMAKE_CURRENT_BINARY_DIR}/maintenance.py"
    KX_BUILD_BINARY="${CMAKE_CURRENT_BINARY_DIR}/kernoryx"
)
configure_file(scripts/maintenance.py maintenance.py @ONLY)
configure_file(packaging/kernoryxd.service.in kernoryxd.service @ONLY)
configure_file(packaging/uninstall.cmake.in uninstall.cmake @ONLY)
if(BUILD_TESTING)
    foreach(name policy controller health process)
        add_executable(${name}-test tests/${name}.cpp)
        target_link_libraries(${name}-test PRIVATE kernoryx_core)
        add_test(NAME ${name} COMMAND ${name}-test)
    endforeach()
    add_executable(ppd-test src/power.cpp tests/ppd.cpp)
    target_link_libraries(ppd-test PRIVATE kernoryx_system)
    target_compile_definitions(ppd-test PRIVATE KX_POWERPROFILESCTL="/usr/bin/true")
    add_test(NAME delegated-power COMMAND ppd-test)
    add_test(NAME integration COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py ${CMAKE_CURRENT_BINARY_DIR})
    set_tests_properties(integration PROPERTIES SKIP_RETURN_CODE 77)
    add_test(NAME maintenance COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/tests/maintenance.py)
endif()
install(TARGETS kernoryx RUNTIME DESTINATION bin)
install(TARGETS kernoryxd RUNTIME DESTINATION sbin)
install(PROGRAMS ${CMAKE_CURRENT_BINARY_DIR}/maintenance.py DESTINATION libexec/kernoryx)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/kernoryxd.service DESTINATION lib/systemd/system)
